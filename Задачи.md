private function fetchImagesByProduct(array $linkedFrameIds): array  
{  
    $select = $this->connection->select()  
        ->from(['mg' => 'catalog_product_entity_media_gallery'], [  
            'id' => 'entity_id',  
            'image' => 'value'  
        ])  
        ->joinInner(  
            ['mgv' => 'catalog_product_entity_media_gallery_value'],  
            'mg.value_id = mgv.value_id',  
            ['store_id', 'position']  
        )  
        ->where('mg.entity_id IN (?)', $linkedFrameIds)  
        ->where('mgv.disabled = 0')  
        ->order(new Zend_Db_Expr("  
        mg.entity_id ASC,        mgv.store_id ASC,        CASE WHEN mgv.position = 2 THEN 0 ELSE 1 END ASC,        mgv.position ASC    "));  
  
    $rows = $this->connection->fetchAll($select);  
  
    $result = [];  
  
    foreach ($rows as $row) {  
        $id = (int)$row['id'];  
        $store = (int)$row['store_id'];  
  
        if (isset($result[$id][$store])) {  
            continue;  
        }  
  
        $file = (string)$row['image'];  
        $src = $this->getMediaUrl(Mage_Catalog_Model_Product::ENTITY, $file);  
  
        $result[$id][$store] = $src;  
    }  
  
    return $result;  
}
