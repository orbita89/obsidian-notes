private function fetchImagesByProduct(array $linkedFrameIds): array  
{  
    $select = $this->connection->select()  
        ->from(['mg' => 'catalog_product_entity_media_gallery'], [  
            'id' => 'entity_id',  
            'image' => 'value'  
        ])  
        ->joinInner(  
            ['mgv' => 'catalog_product_entity_media_gallery_value'],  
            'mg.value_id = mgv.value_id',  
            ['store_id', 'position']  
        )  
        ->where('mg.entity_id IN (?)', $linkedFrameIds)  
        ->where('mgv.disabled = 0')  
        ->order(new Zend_Db_Expr("  
        mg.entity_id ASC,        mgv.store_id ASC,        CASE WHEN mgv.position = 2 THEN 0 ELSE 1 END ASC,        mgv.position ASC    "));  
  
    $rows = $this->connection->fetchAll($select);  
  
    $result = [];  
  
    foreach ($rows as $row) {  
        $id = (int)$row['id'];  
        $store = (int)$row['store_id'];  
  
        if (isset($result[$id][$store])) {  
            continue;  
        }  
  
        $file = (string)$row['image'];  
        $src = $this->getMediaUrl(Mage_Catalog_Model_Product::ENTITY, $file);  
  
        $result[$id][$store] = $src;  
    }  
  
    return $result;  
}

 INNER JOIN `catalog_product_entity_varchar` AS `lf` ON lf.entity_id = p.entity_id AND lf.attribute_id = 712 AND lf.store_id = 0 AND lf.entity_type_id = 4
 LEFT JOIN `catalog_product_entity` AS `lfp` ON lfp.sku = lf.value AND lfp.entity_type_id = 4
 LEFT JOIN `catalog_product_entity_varchar` AS `url` ON url.entity_id = lfp.entity_id AND url.attribute_id = 83 AND url.store_id = 0 AND url.entity_type_id = 4
 LEFT JOIN `catalog_product_entity_int` AS `status` ON status.entity_id = lfp.entity_id AND status.attribute_id = 80 AND status.entity_type_id = 4
 LEFT JOIN (SELECT
                     mg.entity_id AS entity_id,
                     mgv.store_id AS store_id,
                     mg.value AS image
