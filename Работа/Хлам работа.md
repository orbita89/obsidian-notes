/**  
 * @return int[]  
 * @throws GuzzleException|RedirectException|RetriableException|Exception  
 */private function getProductIdsForFilter(): array  
{  
    $productIds = $this->getProductIds();  
    $externalSearchProductIds = $this->getProductIdsFromExternalSearch();  
  
    if (!$productIds || $externalSearchProductIds === null) {  
        return $productIds;  
    }  
  
    return array_intersect($productIds, $externalSearchProductIds)  
        ?: SearchResultInterface::NOT_FOUND_IDS;  
}
