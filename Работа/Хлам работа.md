public function testSearch(): void

{

$searchIndexMock = $this->createMock(SearchIndex::class);

$searchIndexMock

->expects($this->any())

->method('search')

->willReturn($this->getAlgoliaResponse())

;

$client = $this->createMock(SearchClient::class);

$client

->expects($this->any())

->method('initIndex')

->willReturnCallback(

function ($index) use ($searchIndexMock) {

if ($index == $_ENV['ALGOLIA_REDIRECT_INDEX']) {

$redirectMock = $this->createMock(SearchIndex::class);

$redirectMock->method('search')

->willReturn([]);

return $redirectMock;

}

return $searchIndexMock;

}

)

;

$dispatcher = $this->createMock(EventDispatcher::class);

$algoliaIndexService = $this->createMock(AlgoliaIndexService::class);

$service = new AlgoliaSearchService(

$client,

$dispatcher,

$algoliaIndexService,

);

$algoliaResult = $service->search('test query', WebsiteId::GlassesUSA);

$this->assertInstanceOf(AlgoliaSearchResult::class, $algoliaResult);

}
