/**  
 * @return int[]  
 * @throws GuzzleException|RedirectException|RetriableException|Exception  
 */private function getProductIdsForFilter(): array  
{  
    $productIds = $this->getProductIds();  
    $externalSearchProductIds = $this->getProductIdsFromExternalSearch();  
  
    if (!$productIds || $externalSearchProductIds === null) {  
        return $productIds;  
    }  
  
    return array_intersect($productIds, $externalSearchProductIds)  
        ?: SearchResultInterface::NOT_FOUND_IDS;  
}

public function testProductsNoSearchResultsException(): void  
{  
    $storeId = self::STORE_ID;  
  
    $productRepository = $this->createMock(\App\Repository\ProductRepository::class);  
    $productRepository  
        ->method('getAll')  
        ->willThrowException(new NoSearchResultsException());  
  
    static::getContainer()->set('test.' . \App\Repository\ProductRepository::class, $productRepository);  
  
    $this->client->request(  
        'GET',  
        "{$this->getVersionUri()}/{$storeId}/products/123", // укажи categoryId  
        ['q' => 'any'] // параметры для GET  
    );  
  
    /** @var JsonResponse $response */  
    $response = $this->client->getResponse();  
  
    $this->assertInstanceOf(JsonResponse::class, $response);  
    $this->assertEquals(Response::HTTP_OK, $response->getStatusCode());  
  
    $content = json_decode($response->getContent(), true);  
    $this->assertEquals(['total' => 0, 'items' => []], $content);  
}
