```dataviewjs
// ==== График Настроя и Энергии ====
const notes = dv.pages('"Ежедневник"')
    .where(p => p.дата)
    .sort(p => p.дата, 'asc');

const labels = notes.map(n => n.дата);
const mood = notes.map(n => n.настрой ?? 0);
const energy = notes.map(n => n.энергия ?? 0);

let chartBlock = `
\`\`\`chart
type: line
labels: ["${labels.join('","')}"]
datasets:
  - name: Настрой
    values: [${mood.join(',')}]
  - name: Энергия
    values: [${energy.join(',')}]
options:
  title: "Настрой и энергия"
  legend: true
  xAxis:
    title: "Дата"
  yAxis:
    title: "Баллы"
    min: 0
    max: 10
\`\`\`
`;

dv.paragraph(chartBlock);


```
