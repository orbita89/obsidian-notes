```dataviewjs
// ==== Настрой и Энергия — график с заливкой под линиями ====
const notes = dv.pages('"Ежедневник"')
    .where(p => p.дата && p.настрой != null && p.энергия != null)
    .sort(p => p.дата, 'asc');

if (notes.length === 0) {
    dv.paragraph("Нет данных для построения графика.");
} else {
    // Группируем по неделям года
    const weeksMap = new Map();

    notes.forEach(n => {
        const dt = dv.luxon.DateTime.fromISO(n.дата);
        const week = `${dt.weekYear}-W${dt.weekNumber}`;
        if (!weeksMap.has(week)) weeksMap.set(week, []);
        weeksMap.get(week).push({ mood: Number(n.настрой), energy: Number(n.энергия) });
    });

    // Формируем массивы для графика
    const labels = [];
    const moodAvg = [];
    const energyAvg = [];

    for (let [week, entries] of Array.from(weeksMap.entries()).sort()) {
        labels.push(week);
        const moodSum = entries.reduce((acc, e) => acc + e.mood, 0);
        const energySum = entries.reduce((acc, e) => acc + e.energy, 0);
        moodAvg.push((moodSum / entries.length).toFixed(1));
        energyAvg.push((energySum / entries.length).toFixed(1));
    }

    // Создаём Chart с заливкой под линиями
    dv.el("div", `
\`\`\`chart
type: line
labels: ["${labels.join('","')}"]
series:
  - name: Настрой
    data: [${moodAvg.join(',')}]
    color: "#1f77b4"
    showPoints: true
    fill: true
    fillColor: "rgba(31, 119, 180, 0.2)"
  - name: Энергия
    data: [${energyAvg.join(',')}]
    color: "#2ca02c"
    showPoints: true
    fill: true
    fillColor: "rgba(44, 160, 44, 0.2)"
options:
  title: "Настрой и энергия (по неделям)"
  legend: true
  xAxis:
    title: "Неделя"
    ticks:
      maxRotation: 45
      minRotation: 45
      autoSkip: true
      maxTicksLimit: 30
  yAxis:
    title: "Баллы"
    min: 0
    max: 10
\`\`\`
`, { cls: "markdown" });
}
```


### 🔹 Что делает этот ГРАФИК:

1. Берёт все записи из `Ежедневник`.
    
2. Вычисляет **номер недели в году** для каждой даты.
    
3. Группирует `настрой` и `энергию` по неделям и берёт **среднее значение** за неделю.
    
4. Строит график, где по оси X — недели, по Y — баллы от 0 до 10.
