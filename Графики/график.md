// ==== Продвинутый график Настроя и Энергии по неделям ====
const notes = dv.pages('"Ежедневник"')
    .where(p => p.дата && p.настрой != null && p.энергия != null)
    .sort(p => p.дата, 'asc');

if (notes.length === 0) {
    dv.paragraph("Нет данных для построения графика.");
} else {
    // Функция для получения номера недели в году
    function getWeekNumber(dateString) {
        const d = new Date(dateString);
        const onejan = new Date(d.getFullYear(),0,1);
        return Math.ceil((((d - onejan) / 86400000) + onejan.getDay()+1)/7);
    }

    // Группировка по неделям
    const weeksMap = new Map();
    notes.forEach(n => {
        const year = new Date(n.дата).getFullYear();
        const week = getWeekNumber(n.дата);
        const key = `${year}-W${week}`;
        if (!weeksMap.has(key)) weeksMap.set(key, { mood: [], energy: [] });
        weeksMap.get(key).mood.push(Number(n.настрой));
        weeksMap.get(key).energy.push(Number(n.энергия));
    });

    // Подготовка данных для графика
    const weekLabels = Array.from(weeksMap.keys());
    const weekMood = Array.from(weeksMap.values()).map(v => 
        Math.round(v.mood.reduce((a,b)=>a+b,0)/v.mood.length*10)/10
    );
    const weekEnergy = Array.from(weeksMap.values()).map(v => 
        Math.round(v.energy.reduce((a,b)=>a+b,0)/v.energy.length*10)/10
    );

    // Построение блока Chart
    dv.el("div", `
\`\`\`chart
type: line
labels: ["${weekLabels.join('","')}"]
series:
  - name: Настрой
    data: [${weekMood.join(',')}]
  - name: Энергия
    data: [${weekEnergy.join(',')}]
options:
  title: "Настрой и энергия (по неделям)"
  legend: true
  xAxis:
    title: "Неделя"
    ticks:
      maxRotation: 45
      minRotation: 45
      autoSkip: true
      maxTicksLimit: 20
  yAxis:
    title: "Баллы"
    min: 0
    max: 10
\`\`\`
`, { cls: "markdown" });
}
