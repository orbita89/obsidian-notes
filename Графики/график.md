```dataviewjs
// ==== Настрой и Энергия — умный цветной график по неделям ====
const notes = dv.pages('"Ежедневник"')
    .where(p => p.дата && p.настрой != null && p.энергия != null)
    .sort(p => p.дата, 'asc');

if (notes.length === 0) {
    dv.paragraph("Нет данных для построения графика.");
} else {

    // --- Группировка по неделям ---
    const weeksMap = new Map();

    notes.forEach(n => {
        const dt = dv.luxon.DateTime.fromISO(n.дата);
        const week = `${dt.weekYear}-W${String(dt.weekNumber).padStart(2,'0')}`;
        if (!weeksMap.has(week)) weeksMap.set(week, []);
        weeksMap.get(week).push({
            mood: Number(n.настрой),
            energy: Number(n.энергия)
        });
    });

    const labels = [];
    const moodAvg = [];
    const energyAvg = [];

    for (let [week, entries] of Array.from(weeksMap.entries()).sort()) {
        labels.push(week);
        moodAvg.push(entries.reduce((a,e)=>a+e.mood,0)/entries.length);
        energyAvg.push(entries.reduce((a,e)=>a+e.energy,0)/entries.length);
    }

    // --- Функция выбора цвета по уровню ---
    function getColor(value) {
        if (value <= 4) return "#d9534f";      // красный
        if (value <= 7) return "#f0ad4e";      // жёлтый
        return "#5cb85c";                      // зелёный
    }

    const moodColors = moodAvg.map(v => getColor(v));
    const energyColors = energyAvg.map(v => getColor(v));

    dv.el("div", `
\`\`\`chart
type: line
labels: ["${labels.join('","')}"]
series:
  - name: Настрой
    data: [${moodAvg.map(v=>v.toFixed(1)).join(',')}]
    pointBackgroundColor: [${moodColors.map(c=>`"${c}"`).join(',')}]
    borderColor: "#1f77b4"
    fill: true
    tension: 0.3
  - name: Энергия
    data: [${energyAvg.map(v=>v.toFixed(1)).join(',')}]
    pointBackgroundColor: [${energyColors.map(c=>`"${c}"`).join(',')}]
    borderColor: "#2ca02c"
    fill: true
    tension: 0.3
options:
  title: "Настрой и энергия по неделям"
  legend: true
  tooltip: true
  xAxis:
    title: "Недели года"
    ticks:
      autoSkip: true
      maxTicksLimit: 25
  yAxis:
    title: "Средний балл"
    min: 0
    max: 10
\`\`\`
`, { cls: "markdown" });

}


```


### 🔹 Что делает этот ГРАФИК:

1. Берёт все записи из `Ежедневник`.
    
2. Вычисляет **номер недели в году** для каждой даты.
    
3. Группирует `настрой` и `энергию` по неделям и берёт **среднее значение** за неделю.
    
4. Строит график, где по оси X — недели, по Y — баллы от 0 до 10.
