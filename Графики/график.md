```dataviewjs
// Получаем все заметки из папки "Ежедневник" с полем "дата"
const notes = dv.pages('"Ежедневник"')
    .where(p => p.дата)
    .sort(p => p.дата, 'asc');

// Формируем массивы для графика
const labels = notes.map(n => n.дата);
const mood = notes.map(n => n.настрой ?? 0);
const energy = notes.map(n => n.энергия ?? 0);

// Создаём Chart блок динамически
dv.paragraph(`\`\`\`chart
type: line
labels: ["${labels.join('","')}"]
series:
  - name: Настрой
    data: [${mood.join(',')}]
  - name: Энергия
    data: [${energy.join(',')}]
options:
  title: "Настрой и энергия"
  legend: true
  xAxis:
    title: "Дата"
  yAxis:
    title: "Баллы"
    min: 0
    max: 10
\`\`\``);
```
