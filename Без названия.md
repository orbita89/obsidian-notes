# Переключения контейнеров. Общая функция запуск docker
# ✅ Работа удобный запуск docker
# Переключение контейнеров по профилям
typeset -A PROFILES
PROFILES[ma]="php82 phpma"
PROFILES[elastic]="php84 elasticsearch"

# Кэш статусов контейнеров
typeset -A CONTAINER_STATUS

# --- Загрузка статусов контейнеров ---
load_statuses() {
    local name cont_status

    while IFS= read -r line; do
        name=${line%%:*}
        cont_status=${line#*:}

        # Нормализуем статус
        if [[ "$cont_status" == Up* ]]; then
            CONTAINER_STATUS["$name"]="running"
        elif [[ "$cont_status" == Exited* ]]; then
            CONTAINER_STATUS["$name"]="exited"
        else
            CONTAINER_STATUS["$name"]="none"
        fi
    done < <(docker ps -a --format '{{.Names}}:{{.Status}}')
}

# --- Проверка ---
is_running() {
    [[ "${CONTAINER_STATUS[$1]}" == "running" ]]
}

start_container() {
    local name="$1"

    if ! is_running "$name"; then
        echo "▶ Starting $name..."
        docker start "$name" >/dev/null
        CONTAINER_STATUS["$name"]="running"
    else
        echo "✔ $name already running"
    fi
}

stop_container() {
    local name="$1"

    if is_running "$name"; then
        echo "✖ Stopping $name..."
        docker stop "$name" >/dev/null
        CONTAINER_STATUS["$name"]="exited"
    fi
}

# --- Основная функция ---
dup_profile() {
    local profile="$1"
    local needed_containers=(${=PROFILES[$profile]})

    if [[ -z "$needed_containers" ]]; then
        echo "Ошибка: профиль '$profile' не существует"
        return 1
    fi

    echo "Активируем профиль: $profile"
    echo "Контейнеры должны быть включены: ${needed_containers[@]}"
    echo ""

    # Загружаем кеш статусов
    load_statuses

    # Все известные контейнеры
    local all_containers=("${(k)CONTAINER_STATUS}")

    # Включаем необходимые
    for c in "${needed_containers[@]}"; do
        start_container "$c"
    done

    echo ""
    echo "Проверяем лишние контейнеры…"
    echo ""
    
    

    # Останавливаем лишние
    for c in "${all_containers[@]}"; do
        if [[ ! " ${needed_containers[@]} " =~ " $c " ]]; then
            stop_container "$c"
        fi
    done

    echo ""
    echo "Готово."
}

alias dup-ma='dup_profile ma'
alias dup-elastic='dup_profile elastic'
