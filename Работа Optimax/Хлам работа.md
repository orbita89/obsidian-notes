public function jsonSerialize(): array  
{  
    return [  
        'videoTypeId' => $this->videoTypeId,  
        'genderId' => $this->videoTypeId,  
        'thumbnail' => $this->thumbnail,  
        'videoUrl' => $this->videoUrl,  
        'isDefault' => $this->isDefault,  
        'preview' => $this->preview,  
        'isActiveVideo' => $this->isActiveVideo,  
        'alias' => $this->alias,  
    ];  
}  
  
public static function fromArray(array $data): self  
{  
    return new self(  
        videoTypeId: $data['videoTypeId'] ?? $data['genderId'] ?? 0,  
        thumbnail: (string)($data['thumbnail'] ?? ''),  
        videoUrl: (string)($data['videoUrl'] ?? ''),  
        isDefault: (bool)($data['isDefault'] ?? false),  
        preview: (string)($data['preview'] ?? ''),  
        isActiveVideo: (bool)($data['isActiveVideo'] ?? true),  
        alias: (string)($data['alias'] ?? ''),  
    );  
}
