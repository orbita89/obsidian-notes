<%*
const folder = "Ежедневник";
const filename = `Ежедневник-${tp.date.now("YYYY-MM-DD-HH-mm")}`;

// ==== Запрос данных ====
let important = await tp.system.prompt("📌 Что сегодня было важно?");
let win = await tp.system.prompt("🏆 Маленькая победа?");
let problem = await tp.system.prompt("🚧 Проблема или затык?");
let fix = await tp.system.prompt("🔧 Что попробуете изменить?");
let mood = await tp.system.prompt("Настрой (1–10)");
let energy = await tp.system.prompt("Энергия (1–10)");

if (!/^[1-9]$|^10$/.test(mood)) mood = "не указано";
if (!/^[1-9]$|^10$/.test(energy)) energy = "не указано";

// ==== Формируем контент с YAML сразу ====
let yaml = `---
дата: ${tp.date.now("YYYY-MM-DD")}
важное: |1
  ${important.replace(/\n/g, "\n  ")}
победа: ${win}
проблема: ${problem}
решение: ${fix}
настрой: ${mood}
энергия: ${energy}
---`;

let content = `${yaml}

## 📌 Что сегодня было важно
${important}

## 🚧 Проблема или затык
**Что случилось:**  
${problem}

**Что попробую изменить:**  
${fix}

## 🏆 Маленькая победа
${win}

## 📊 Состояние
Настрой: ${mood}  
Энергия: ${energy}
`;

// ==== Создаём файл в папке Ежедневник ====
await tp.file.create_new(content, filename, false, folder);

// ==== Удаляем текущий временный файл (если был) ====
await tp.file.delete(tp.file.path(true));
%>
