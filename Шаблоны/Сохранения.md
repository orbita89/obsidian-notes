<%*
// ==== Получаем содержимое текущего файла ====
let content = tp.file.content;

// ==== Функция для вытаскивания текста между заголовками ====
function extractSection(text, header) {
  let regex = new RegExp(`## ${header}[\\r\\n]+([\\s\\S]*?)(?=(\\n## |$))`, "m");
  let match = text.match(regex);
  return match ? match[1].trim() : "";
}

// ==== Вытаскиваем все поля ====
let important = extractSection(content, "📌 Что сегодня было важно");
let problemBlock = extractSection(content, "🚧 Проблема или затык");
let problemMatch = problemBlock.match(/\*\*Что случилось:\*\*\s*([\s\S]*?)\n\n/);
let fixMatch = problemBlock.match(/\*\*Что попробую изменить:\*\*\s*([\s\S]*)/);
let problem = problemMatch ? problemMatch[1].trim() : "";
let fix = fixMatch ? fixMatch[1].trim() : "";
let win = extractSection(content, "🏆 Маленькая победа");
let stateBlock = extractSection(content, "📊 Состояние");
let moodMatch = stateBlock.match(/Настрой:\s*(.*)/);
let energyMatch = stateBlock.match(/Энергия:\s*(.*)/);
let mood = moodMatch ? moodMatch[1].trim() : "не указано";
let energy = energyMatch ? energyMatch[1].trim() : "не указано";

// ==== Перезаписываем YAML ====
let yaml = `---
дата: ${tp.date.now("YYYY-MM-DD")}
важное: |
  ${important.replace(/\n/g,"\n  ")}
победа: ${win}
проблема: ${problem}
решение: ${fix}
настрой: ${mood}
энергия: ${energy}
---`;

// ==== Вставляем YAML в файл, оставляя тело без изменений ====
let rest = content.replace(/^---[\s\S]*?---/, "").trim();
tR = `${yaml}\n${rest}`;
%>
