<%*
// ===== Читаем текущий файл =====
let fileContent = await tp.file.read();

// ===== Функция для получения содержимого блока =====
function getBlock(name) {
  let regex = new RegExp(`##\\s*${name}[\\s\\S]*?(?=##|$)`, 'm');
  let match = fileContent.match(regex);
  if (!match) return "";
  return match[0].replace(/^##.*\n/, '').trim();
}

// ===== Получаем данные из Markdown =====
let important = getBlock("📌 Что сегодня было важно");
let win = getBlock("🏆 Маленькая победа");
let problemMatch = getBlock("🚧 Проблема или затык");
let problem = problemMatch.match(/\*\*Что случилось:\*\*\s*([\s\S]*?)\n\n\*\*Что попробую изменить:\*\*/)?.[1]?.trim() || "";
let fix = problemMatch.match(/\*\*Что попробую изменить:\*\*\s*([\s\S]*)/)?.[1]?.trim() || "";
let moodMatch = fileContent.match(/Настрой:\s*(.*)/);
let energyMatch = fileContent.match(/Энергия:\s*(.*)/);
let mood = moodMatch ? moodMatch[1].trim() : "не указано";
let energy = energyMatch ? energyMatch[1].trim() : "не указано";

// ===== Формируем новый YAML =====
let yaml = `---
дата: ${tp.date.now("YYYY-MM-DD")}
важное: |
  ${important.replace(/\n/g, '\n  ')}
победа: ${win}
проблема: ${problem}
решение: ${fix}
настрой: ${mood}
энергия: ${energy}
---`;

// ===== Обновляем YAML в файле =====
fileContent = fileContent.replace(/---[\s\S]*?---/, yaml);

// ===== Перезаписываем файл =====
await tp.file.overwrite(fileContent);
%>
